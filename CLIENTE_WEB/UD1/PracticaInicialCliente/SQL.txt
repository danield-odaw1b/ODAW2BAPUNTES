CREATE TABLE CURSOS (
    ID int,
    clase varchar(20),
    Num_alumnos int,
    PRIMARY KEY(ID)
);

CREATE TABLE ALUMNOS (
    ID int,
    nombre varchar(20),
    apellidos varchar(50),
    ID_curso int,
    PRIMARY KEY(ID),
    FOREIGN KEY(ID_curso) REFERENCES CURSOS(ID)
);

CREATE TABLE PROFESORES (
    ID int,
    nombre varchar(20),
    apellidos varchar(50),
    materia varchar(50),
    ID_curso int,
    PRIMARY KEY(ID),
    FOREIGN KEY(ID_curso) REFERENCES CURSOS(ID)
);
